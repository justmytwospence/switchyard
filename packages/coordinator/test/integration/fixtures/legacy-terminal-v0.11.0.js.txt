var LineSplitter = class {
  constructor(onLine) {
    this.onLine = onLine;
  }
  onLine;
  buffer = "";
  push(text) {
    const buf = this.buffer + text;
    let start = 0;
    for (let nl = buf.indexOf("\n"); nl >= 0; nl = buf.indexOf("\n", start)) {
      this.emit(buf.slice(start, nl));
      start = nl + 1;
    }
    this.buffer = buf.slice(start);
  }
  /** Return and clear the unterminated rest (CR stripped). */
  end() {
    const rest = this.buffer;
    this.buffer = "";
    return rest.endsWith("\r") ? rest.slice(0, -1) : rest;
  }
  emit(raw) {
    const line = raw.endsWith("\r") ? raw.slice(0, -1) : raw;
    if (line) this.onLine(line);
  }
};

  "low",
  "medium",

function supportedThinkingLevels(model) {
  const known = PI_THINKING_LEVELS;
  if (Array.isArray(model?.thinkingLevels)) {
    return model.thinkingLevels.filter((l) => typeof l === "string" && known.includes(l));
  }
  if (!model?.reasoning) return ["off"];
  return PI_THINKING_LEVELS.filter((level) => {
    const mapped = model.thinkingLevelMap?.[level];
    if (mapped === null) return false;
    if (level === "xhigh" || level === "max") return mapped !== void 0;
    return true;
  });
}

      if (socket === s) socket = null;
      if (!connected && attempt < CONNECT_ATTEMPTS) {
        setTimeout(() => connect(attempt + 1), CONNECT_RETRY_MS).unref();
      }
    });
    s.unref();
    s.write(`${JSON.stringify({ role: "extension" })}
`);
  };
  connect(1);
  try {
    pi.events?.on?.(REMOTE_UI_CHANNEL, (data) => {
      const r = data;
      if (r?.version !== 1 || r.ui || !socket) return;
      r.ui = remoteUi;
      r.provider = "switchyard";
    });
  } catch {
  }
  const on = (name, fn) => {
    try {
      pi.on(name, (event, c) => {
        try {
          fn(event, c);
        } catch {
        }
      });
    } catch {
    }
  };
  on("session_start", (event, c) => {
    ctx = c;
    patchUi(c);
    completer = null;
    try {
      c.ui?.addAutocompleteProvider?.((current) => {
        completer = current;
        return current;
      });
    } catch {
    }
    entryCount = c.sessionManager.getEntries().length;
    const sessionId = c.sessionManager.getSessionId();
    const sessionFile = c.sessionManager.getSessionFile() ?? null;
    hello();
    sendModelState();
    sendLeaf();
    if (event?.reason && event.reason !== "startup") {
      send({ type: "switchyard_native", sessionId, sessionFile });
    }
    for (const cmd of early.splice(0)) handleCommand(cmd);
  });
  for (const name of [
    "agent_start",
    "message_start",
    "tool_execution_start",
    "tool_execution_update",
    "tool_execution_end",
    "session_info_changed"
  ]) {
    on(name, (event) => send(event));
  }
  on("message_start", (event) => {
    if (event?.message?.role !== "user") return;
    const text = messageText(event.message);
    for (const list of [queue.steering, queue.followUp]) {
      const i = list.indexOf(text);
      if (i >= 0) {
        list.splice(i, 1);
        sendQueue();
        return;
      }
    }
  });
  on("message_update", (event) => send(jsonMessageUpdate(event)));
  on("message_end", (event) => {
    send(event);
    flushEntries();
  });
  on("turn_end", () => {
    flushEntries();
    sendLeaf();
  });
  on("session_tree", () => {
    flushEntries();
    sendLeaf();
  });
  on("agent_settled", (event) => {
    send({ type: "agent_settled", aborted: event?.aborted === true });
    flushEntries();
    sendLeaf();
    if (queue.steering.length > 0 || queue.followUp.length > 0) {
      queue.steering = [];
      queue.followUp = [];
      sendQueue();
    }
  });
  on("session_compact", () => {
    send({ type: "compaction_end", result: true });
    flushEntries();
  });
  on(
    "session_compact_failed",
    (event) => send({
      type: "compaction_end",
      result: null,
      aborted: event?.aborted === true,
      errorMessage: event?.errorMessage ?? "unknown error"
    })
  );
  on("model_select", () => sendModelState());
  on("thinking_level_select", () => sendModelState());
  on(
    "session_before_compact",
    (event) => send({ type: "compaction_start", reason: event?.reason ?? "manual" })
  );
  on(
    "ui_prompt_start",
    (event) => send({
      type: "ui_prompt",
      event: event.type,
      kind: event.kind,
      title: event.title ?? null,
      remoteBacked: event.kind === "custom" && remoteBacked > 0
    })
  );
  on(
    "ui_prompt_end",
    (event) => send({ type: "ui_prompt", event: event.type, kind: event.kind, title: event.title ?? null })
  );
  on("input", (event) => {
    if (event?.source !== "interactive" || !event.streamingBehavior) return;
    (event.streamingBehavior === "steer" ? queue.steering : queue.followUp).push(
      String(event.text ?? "")
    );
    sendQueue();
  });
}
